  <script type="module">
    // Use CDNs that provide pre-bundled modules to avoid internal dependency issues
    let THREE, GLTFLoader, OrbitControls;
    
    async function loadThreeJS() {
      try {
        // Load from skypack.dev which handles module dependencies better
        THREE = await import('https://cdn.skypack.dev/three@0.161.0');
        GLTFLoader = await import('https://cdn.skypack.dev/three@0.161.0/examples/jsm/loaders/GLTFLoader.js');
        OrbitControls = await import('https://cdn.skypack.dev/three@0.161.0/examples/jsm/controls/OrbitControls.js');
        
        console.log('All modules loaded successfully from Skypack');
        
        // Extract the classes from the modules
        GLTFLoader = GLTFLoader.GLTFLoader;
        OrbitControls = OrbitControls.OrbitControls;
        
        // Start the application
        initApp();
        
      } catch (e) {
        console.warn('Skypack failed, trying alternative CDN:', e);
        
        try {
          // Fallback to esm.sh which also handles dependencies well
          THREE = await import('https://esm.sh/three@0.161.0');
          GLTFLoader = await import('https://esm.sh/three@0.161.0/examples/jsm/loaders/GLTFLoader.js');
          OrbitControls = await import('https://esm.sh/three@0.161.0/examples/jsm/controls/OrbitControls.js');
          
          console.log('All modules loaded successfully from ESM.sh');
          
          // Extract the classes from the modules
          GLTFLoader = GLTFLoader.GLTFLoader;
          OrbitControls = OrbitControls.OrbitControls;
          
          // Start the application
          initApp();
          
        } catch (e2) {
          console.error('All CDNs failed:', e2);
          document.getElementById('readout').textContent = 'Error: Failed to load Three.js. Please check your internet connection and refresh the page.';
        }
      }
    }
    
    // Initialize the application after loading Three.js
    function initApp() {
      // Core scene setup
      const app = document.getElementById('app');
      const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
      renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
      renderer.setSize(innerWidth, innerHeight);
      renderer.outputColorSpace = THREE.SRGBColorSpace;
      renderer.shadowMap.enabled = true;
      renderer.shadowMap.type = THREE.PCFSoftShadowMap;
      app.appendChild(renderer.domElement);

      const scene = new THREE.Scene();
      scene.background = new THREE.Color(0x87CEEB); // Bright sky blue instead of dark blue
      const camera = new THREE.PerspectiveCamera(60, innerWidth/innerHeight, 0.1, 5000);
      camera.position.set(0, 3, 8);

      // Lights - brighter and more vibrant
      const hemi = new THREE.HemisphereLight(0xE0F6FF, 0x87CEEB, 1.2); // Brighter sky light, more intense
      scene.add(hemi);
      const dirLight = new THREE.DirectionalLight(0xFFFFFF, 1.5); // Brighter directional light
      dirLight.position.set(60, 100, 30);
      dirLight.castShadow = true;
      dirLight.shadow.mapSize.set(2048, 2048);
      dirLight.shadow.camera.near = 1;
      dirLight.shadow.camera.far = 400;
      dirLight.shadow.camera.left = -120;
      dirLight.shadow.camera.right = 120;
      dirLight.shadow.camera.top = 120;
      dirLight.shadow.camera.bottom = -120;
      scene.add(dirLight);

      // Ground - lighter to match brighter sky
      const groundGeo = new THREE.PlaneGeometry(2000, 2000);
      const groundMat = new THREE.MeshStandardMaterial({ color: 0x90EE90, metalness: 0, roughness: 1 }); // Light green ground
      const ground = new THREE.Mesh(groundGeo, groundMat);
      ground.rotation.x = -Math.PI/2;
      ground.receiveShadow = true;
      ground.position.y = -0.01;
      scene.add(ground);

      // Loaders
      const loader = new GLTFLoader();

      // State
      const mixers = new Set();
      let orbitControls = null;
      let useOrbit = false;

      // Scene container and current plane wrapper (pivot at model center)
      const sceneRoot = new THREE.Group();
      scene.add(sceneRoot);

      const planePivot = new THREE.Group();
      planePivot.position.set(0, 3, 0);
      scene.add(planePivot);

      // Flight model
      const input = {
        pitch: 0, yaw: 0, roll: 0, throttle: 0.5,
        up: false, down: false, left: false, right: false, rollL: false, rollR: false, boost: false, brake: false
      };
      const flight = {
        speedMin: 6,
        speedMax: 120,
        speed: 0,
        accel: 18,
        decel: 24,
        turnRate: THREE.MathUtils.degToRad(60),   // yaw per second at full input
        pitchRate: THREE.MathUtils.degToRad(55),  // pitch per second
        rollRate: THREE.MathUtils.degToRad(110),  // roll per second
        drag: 0.0012,                             // mild quadratic drag
        autoLevel: THREE.MathUtils.degToRad(22),  // roll leveling per second
      };

      // Camera follow
      const camRig = new THREE.Group();
      const camTarget = new THREE.Object3D();
      planePivot.add(camRig);
      planePivot.add(camTarget);
      camTarget.position.set(0, 1.2, -6);
      camRig.position.set(0, 2.0, 8.0); // will be adjusted by slider
      camRig.add(camera);

      // HUD elements
      const readout = document.getElementById('readout');
      const speedKV = document.getElementById('speedKV');
      const throttleKV = document.getElementById('throttleKV');
      const fpsKV = document.getElementById('fpsKV');

      // UI elements
      const sceneFile = document.getElementById('sceneFile');
      const planeFile = document.getElementById('planeFile');
      const planeSelect = document.getElementById('planeSelect');
      const toggleOrbitBtn = document.getElementById('toggleOrbit');
      const camDistance = document.getElementById('camDistance');

      // Helper: load GLB from URL or object URL, return group with centered pivot and fitted scale
      async function loadGLB(urlOrBlobURL) {
        const gltf = await loader.loadAsync(urlOrBlobURL);
        const root = gltf.scene || gltf.scenes[0];
        root.traverse((o) => {
          if (o.isMesh) {
            o.castShadow = true;
            o.receiveShadow = true;
            if (o.material && o.material.map) o.material.map.anisotropy = 8;
          }
        });

        // Center and get size
        const box = new THREE.Box3().setFromObject(root);
        const size = new THREE.Vector3();
        box.getSize(size);
        const center = new THREE.Vector3();
        box.getCenter(center);

        const pivot = new THREE.Group();
        // Reparent with centered pivot
        root.position.sub(center);
        pivot.add(root);

        // Auto-scale extremely large/small assets to a friendly size (approximate 5 unit width)
        const target = 5;
        const largest = Math.max(size.x, size.y, size.z) || 1;
        const scale = THREE.MathUtils.clamp(target / largest, 0.05, 20);
        pivot.scale.setScalar(scale);

        return { pivot, gltf };
      }

      // Asset management
      let currentPlane = null;
      let currentPlaneMixer = null;

      async function setPlaneFromURL(url, options = { tryAltFor7: false }) {
        try {
          const { pivot, gltf } = await loadGLB(url);
          swapPlane(pivot, gltf);
        } catch (e) {
          // Fallback for plane7.blg typo case
          if (options.tryAltFor7 && url.endsWith('plane7.glb')) {
            try {
              const { pivot, gltf } = await loadGLB('plane7.blg');
              swapPlane(pivot, gltf);
              return;
            } catch (e2) {
              console.warn('Failed loading plane7.blg fallback:', e2);
              console.error('Original error:', e);
            }
          } else {
            console.error('Failed to load plane:', url, e);
          }
        }
      }

      function swapPlane(pivot, gltf) {
        // Remove previous
        if (currentPlane) {
          planePivot.remove(currentPlane);
          currentPlane.traverse(o => { if (o.geometry) o.geometry.dispose?.(); if (o.material) o.material.dispose?.(); });
          currentPlane = null;
        }
        if (currentPlaneMixer) {
          mixers.delete(currentPlaneMixer);
          currentPlaneMixer.uncacheRoot(currentPlaneMixer.getRoot());
          currentPlaneMixer = null;
        }

        // Attach
        pivot.rotation.set(0, Math.PI, 0); // face forward (-Z) common convention
        planePivot.add(pivot);
        currentPlane = pivot;

        // Play first animation if present (e.g., propeller)
        if (gltf.animations && gltf.animations.length) {
          const mixer = new THREE.AnimationMixer(pivot);
          const action = mixer.clipAction(gltf.animations[0]);
          action.play();
          mixers.add(mixer);
          currentPlaneMixer = mixer;
        }

        // Reset kinematics on swap, keep throttle
        planePivot.position.set(0, 3, 0);
        planePivot.rotation.set(0, 0, 0);
        flight.speed = THREE.MathUtils.lerp(flight.speedMin, flight.speedMax, input.throttle);

        // Update UI
        readout.textContent = `Plane loaded: ✓  (tris will vary)
Keys: W/S pitch | A/D yaw | Q/E roll | Shift/Ctrl throttle | 1–7 switch | R reset`;
      }

      let currentScene = null;
      async function setSceneFromURL(url) {
        try {
          const { pivot } = await loadGLB(url);
          // Cleanup old scene
          if (currentScene) {
            sceneRoot.remove(currentScene);
            currentScene.traverse(o => { if (o.geometry) o.geometry.dispose?.(); if (o.material) o.material.dispose?.(); });
          }
          // Scenes usually should not be auto-scaled to tiny sizes: relax scaling rule
          // But loadGLB already normalized sizes; so if scene has large terrain, you may re-scale:
          pivot.scale.multiplyScalar(10);
          sceneRoot.add(pivot);
          currentScene = pivot;
        } catch (e) {
          console.error('Failed to load scene:', e);
        }
      }

      // File inputs
      sceneFile.addEventListener('change', (e) => {
        const f = e.target.files?.[0];
        if (f) setSceneFromURL(URL.createObjectURL(f));
      });
      planeFile.addEventListener('change', (e) => {
        const f = e.target.files?.[0];
        if (f) {
          setPlaneFromURL(URL.createObjectURL(f));
          planeSelect.value = ''; // clear selection
        }
      });
      planeSelect.addEventListener('change', (e) => {
        const v = e.target.value;
        if (!v) return;
        setPlaneFromURL(v, { tryAltFor7: v === 'plane7.glb' });
      });

      // Drag & Drop for quick plane swap
      document.addEventListener('dragover', (e) => { e.preventDefault(); });
      document.addEventListener('drop', (e) => {
        e.preventDefault();
        const f = [...(e.dataTransfer?.files || [])].find(f => /(\.glb|\.gltf)$/i.test(f.name));
        if (f) setPlaneFromURL(URL.createObjectURL(f));
      });

      // Orbit controls toggle (debug)
      toggleOrbitBtn.addEventListener('click', () => {
        useOrbit = !useOrbit;
        if (useOrbit && !orbitControls) {
          orbitControls = new OrbitControls(camera, renderer.domElement);
          orbitControls.enableDamping = true;
        }
        if (orbitControls) {
          orbitControls.enabled = useOrbit;
          if (useOrbit) orbitControls.target.copy(planePivot.position);
        }
        toggleOrbitBtn.textContent = useOrbit ? 'Disable Orbit Controls' : 'Enable Orbit Controls';
      });

      camDistance.addEventListener('input', (e) => {
        const d = parseFloat(e.target.value);
        camRig.position.set(0, d * 0.22 + 1.2, d);
      });

      // Keyboard input
      const key = {};
      function onKey(e, down) {
        key[e.code] = down;
        if (!down) return;

        // Plane quick select
        if (/^Digit[1-7]$/.test(e.code)) {
          const n = e.code.slice(-1);
          const fname = `plane${n}.glb`;
          planeSelect.value = fname;
          setPlaneFromURL(fname, { tryAltFor7: n === '7' });
        }
        if (e.code === 'KeyR') {
          planePivot.position.set(0, 3, 0);
          planePivot.rotation.set(0, 0, 0);
          flight.speed = THREE.MathUtils.lerp(flight.speedMin, flight.speedMax, input.throttle);
        }
      }
      window.addEventListener('keydown', (e) => onKey(e, true));
      window.addEventListener('keyup',   (e) => onKey(e, false));

      function updateInput(dt) {
        // Axes
        input.pitch = (key['KeyW'] ? -1 : 0) + (key['KeyS'] ? 1 : 0);
        input.yaw   = (key['KeyA'] ? 1 : 0) + (key['KeyD'] ? -1 : 0);
        input.roll  = (key['KeyQ'] ? 1 : 0) + (key['KeyE'] ? -1 : 0);
        // Throttle
        if (key['ShiftLeft'] || key['ShiftRight']) input.throttle = Math.min(1, input.throttle + 0.6 * dt);
        if (key['ControlLeft'] || key['ControlRight']) input.throttle = Math.max(0, input.throttle - 0.6 * dt);
      }

      // Initial loads
      setSceneFromURL('scene.glb');                 // Will succeed if present
      setPlaneFromURL('plane1.glb');                // Active control plane
      setPlaneFromURL('plane2.glb');                // Preload in browser cache
      setPlaneFromURL('plane3.glb');                // (no attach)
      setPlaneFromURL('plane4.glb');
      setPlaneFromURL('plane5.glb');
      setPlaneFromURL('plane6.glb');
      setPlaneFromURL('plane7.glb', { tryAltFor7: true });

      // Resize handling
      window.addEventListener('resize', () => {
        camera.aspect = innerWidth/innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(innerWidth, innerHeight);
      });

      // FPS simple tracker
      let frames = 0, timeAcc = 0, fps = 0;

      // Main loop
      const clock = new THREE.Clock();
      function tick() {
        const dt = Math.min(clock.getDelta(), 0.05);

        // Input and throttle -> speed
        updateInput(dt);
        const targetSpeed = THREE.MathUtils.lerp(flight.speedMin, flight.speedMax, input.throttle);
        const accel = targetSpeed > flight.speed ? flight.accel : flight.decel;
        flight.speed += (targetSpeed - flight.speed) * (1 - Math.exp(-accel * dt));
        // Drag (quadratic, mild)
        flight.speed -= flight.drag * flight.speed * flight.speed * dt;
        flight.speed = THREE.MathUtils.clamp(flight.speed, flight.speedMin * 0.5, flight.speedMax * 1.2);

        // Rotations (scaled by speed factor for a nice feel)
        const speedFactor = THREE.MathUtils.clamp((flight.speed - flight.speedMin) / (flight.speedMax - flight.speedMin), 0, 1);
        const pitch = input.pitch * flight.pitchRate * dt * (0.7 + 0.6 * speedFactor);
        const yaw   = input.yaw   * flight.turnRate  * dt * (0.7 + 0.6 * speedFactor);
        const roll  = input.roll  * flight.rollRate  * dt * (0.7 + 0.6 * speedFactor);

        planePivot.rotateX(pitch);
        planePivot.rotateY(yaw);
        planePivot.rotateZ(roll);

        // Gentle auto-level on roll when no roll input
        if (Math.abs(input.roll) < 0.02) {
          const r = planePivot.rotation.z;
          const level = -r * Math.min(1, flight.autoLevel * dt);
          planePivot.rotateZ(level);
        }

        // Advance forward in local -Z
        const forward = new THREE.Vector3(0, 0, -1).applyQuaternion(planePivot.quaternion).normalize();
        planePivot.position.addScaledVector(forward, flight.speed * dt);

        // Prevent tunneling below simple ground plane
        planePivot.position.y = Math.max(planePivot.position.y, 0.5);

        // Camera follow (smooth chase)
        // Desired camera position relative to plane
        const desired = camRig.position.clone();
        desired.applyQuaternion(planePivot.quaternion);
        desired.add(planePivot.position);

        // Smoothly move camera
        camera.position.lerp(desired, 1 - Math.exp(-8 * dt));
        const lookAt = camTarget.position.clone().applyQuaternion(planePivot.quaternion).add(planePivot.position);
        camera.lookAt(lookAt);

        // Orbit controls (debug)
        if (orbitControls && orbitControls.enabled) {
          orbitControls.target.lerp(planePivot.position, 1 - Math.exp(-6 * dt));
          orbitControls.update();
        }

        // Mixers
        mixers.forEach(m => m.update(dt));

        // Lights follow
        dirLight.position.lerp(new THREE.Vector3().copy(planePivot.position).add(new THREE.Vector3(60, 100, 30)), 1 - Math.exp(-1.5 * dt));

        renderer.render(scene, camera);

        // HUD updates
        speedKV.textContent = flight.speed.toFixed(1);
        throttleKV.textContent = Math.round(input.throttle * 100) + '%';
        frames++; timeAcc += dt;
        if (timeAcc >= 0.5) { fps = Math.round(frames / timeAcc); frames = 0; timeAcc = 0; fpsKV.textContent = fps; }

        requestAnimationFrame(tick);
      }
      requestAnimationFrame(tick);
    }

    // Start loading Three.js
    loadThreeJS().catch(error => {
      console.error('Failed to load Three.js:', error);
      document.getElementById('readout').textContent = 'Error: Failed to load Three.js. Please check your internet connection and refresh the page.';
    });
  </script>
