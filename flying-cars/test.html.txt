  <script type="module">
    // Game state
    let gameLoaded = false;
    let currentPlaneIndex = 0;
    const planeFiles = ['plane1.glb', 'plane2.glb', 'plane3.glb', 'plane4.glb', 'plane5.glb', 'plane6.glb'];
    
    // Three.js components
    let THREE, GLTFLoader;
    let scene, camera, renderer;
    let planePivot, currentPlane, currentPlaneMixer;
    let mixers = new Set();
    
         // Flight physics
     let flight;
    
    // Input state
    const input = {
      pitch: 0, yaw: 0, roll: 0, throttle: 0.5
    };
    
    // Camera follow
    let camRig, camTarget;
    
    // Loading progress
    let loadingProgress = 0;
    const totalAssets = planeFiles.length + 1; // +1 for scene
    
    // Update loading bar
    function updateLoadingProgress() {
      loadingProgress++;
      const percentage = (loadingProgress / totalAssets) * 100;
      document.getElementById('loadingProgress').style.width = percentage + '%';
      
      if (loadingProgress >= totalAssets) {
        setTimeout(() => {
          document.getElementById('loadingScreen').classList.add('hidden');
          gameLoaded = true;
        }, 500);
      }
    }
    
         // Load Three.js
     async function loadThreeJS() {
       try {
         // Try esm.sh first (more reliable)
         THREE = await import('https://esm.sh/three@0.158.0');
         GLTFLoader = await import('https://esm.sh/three@0.158.0/examples/jsm/loaders/GLTFLoader.js');
         GLTFLoader = GLTFLoader.GLTFLoader;
         
         console.log('Three.js loaded successfully from ESM.sh');
         initGame();
         
       } catch (e) {
         console.warn('ESM.sh failed, trying alternative CDN:', e);
         
         try {
           // Fallback to unpkg
           THREE = await import('https://unpkg.com/three@0.158.0/build/three.module.js');
           GLTFLoader = await import('https://unpkg.com/three@0.158.0/examples/jsm/loaders/GLTFLoader.js');
           GLTFLoader = GLTFLoader.GLTFLoader;
           
           console.log('Three.js loaded successfully from unpkg');
           initGame();
           
         } catch (e2) {
           console.error('All CDNs failed:', e2);
           document.getElementById('loadingScreen').innerHTML = '<div class="loadingText">Error: Failed to load Three.js from all CDNs. Please check your internet connection and refresh the page.</div>';
         }
       }
     }
    
         // Initialize the game
     function initGame() {
       // Initialize flight physics after Three.js is loaded
       flight = {
         speed: 0,
         speedMin: 8,
         speedMax: 150,
         accel: 20,
         decel: 25,
         turnRate: THREE.MathUtils.degToRad(70),
         pitchRate: THREE.MathUtils.degToRad(60),
         rollRate: THREE.MathUtils.degToRad(120),
         autoLevel: THREE.MathUtils.degToRad(25)
       };
       
       // Scene setup
       scene = new THREE.Scene();
      scene.background = new THREE.Color(0x87CEEB);
      scene.fog = new THREE.Fog(0x87CEEB, 100, 1000);
      
      // Camera setup
      camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 2000);
      camera.position.set(0, 5, 15);
      
             // Renderer setup
       renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
       renderer.setSize(window.innerWidth, window.innerHeight);
       renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
       renderer.shadowMap.enabled = false; // Disabled shadows for better performance
       renderer.outputColorSpace = THREE.SRGBColorSpace;
      document.getElementById('gameContainer').appendChild(renderer.domElement);
      
      // Lighting
      setupLighting();
      
      // Ground and sky
      createEnvironment();
      
      // Flight controls
      setupFlightControls();
      
      // Camera follow system
      setupCameraFollow();
      
      // Load assets
      loadAssets();
      
      // Start game loop
      gameLoop();
      
      // Handle window resize
      window.addEventListener('resize', onWindowResize);
    }
    
         // Setup lighting
     function setupLighting() {
       // Much brighter ambient light to make planes visible
       const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
       scene.add(ambientLight);
       
       // Brighter directional light (sun)
       const directionalLight = new THREE.DirectionalLight(0xffffff, 2.0);
       directionalLight.position.set(100, 100, 50);
       directionalLight.castShadow = false; // Disabled shadows
       scene.add(directionalLight);
       
       // Brighter hemisphere light for sky
       const hemisphereLight = new THREE.HemisphereLight(0x87CEEB, 0xffffff, 1.2);
       scene.add(hemisphereLight);
       
       // Additional fill light from below to eliminate dark areas
       const fillLight = new THREE.DirectionalLight(0xffffff, 0.6);
       fillLight.position.set(0, -50, 0);
       scene.add(fillLight);
     }
    
    // Create environment
    function createEnvironment() {
             // Ground - much brighter to eliminate dark areas
       const groundGeometry = new THREE.PlaneGeometry(2000, 2000);
       const groundMaterial = new THREE.MeshStandardMaterial({ 
         color: 0x98FB98, // Brighter green
         roughness: 0.3,  // Less rough for more brightness
         metalness: 0.0   // No metalness for cleaner look
       });
       const ground = new THREE.Mesh(groundGeometry, groundMaterial);
       ground.rotation.x = -Math.PI / 2;
       ground.position.y = -0.1;
       scene.add(ground);
      
      // Sky dome
      const skyGeometry = new THREE.SphereGeometry(500, 32, 32);
      const skyMaterial = new THREE.MeshBasicMaterial({
        color: 0x87CEEB,
        side: THREE.BackSide
      });
      const sky = new THREE.Mesh(skyGeometry, skyMaterial);
      scene.add(sky);
      
      // Clouds (simple spheres)
      for (let i = 0; i < 20; i++) {
        const cloudGeometry = new THREE.SphereGeometry(Math.random() * 30 + 20, 8, 8);
        const cloudMaterial = new THREE.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.8 });
        const cloud = new THREE.Mesh(cloudGeometry, cloudMaterial);
        cloud.position.set(
          (Math.random() - 0.5) * 800,
          Math.random() * 100 + 50,
          (Math.random() - 0.5) * 800
        );
        scene.add(cloud);
      }
    }
    
    // Setup flight controls
    function setupFlightControls() {
      // Create plane pivot
      planePivot = new THREE.Group();
      planePivot.position.set(0, 10, 0);
      scene.add(planePivot);
      
      // Keyboard controls
      const keyState = {};
      
             window.addEventListener('keydown', (e) => {
         keyState[e.code] = true;
         
         // Only allow controls after game is loaded
         if (!gameLoaded) return;
         
         // Plane switching
         if (e.code >= 'Digit1' && e.code <= 'Digit6') {
           const planeIndex = parseInt(e.code.slice(-1)) - 1;
           if (planeIndex < planeFiles.length) {
             switchPlane(planeIndex);
           }
         }
         
         // Reset position
         if (e.code === 'KeyR') {
           resetPlanePosition();
         }
       });
      
      window.addEventListener('keyup', (e) => {
        keyState[e.code] = false;
      });
      
      // Update input state
      function updateInput() {
        input.pitch = (keyState['KeyW'] ? -1 : 0) + (keyState['KeyS'] ? 1 : 0);
        input.yaw = (keyState['KeyA'] ? 1 : 0) + (keyState['KeyD'] ? -1 : 0);
        input.roll = (keyState['KeyQ'] ? 1 : 0) + (keyState['KeyE'] ? -1 : 0);
        
        if (keyState['ShiftLeft'] || keyState['ShiftRight']) {
          input.throttle = Math.min(1, input.throttle + 0.8 * 0.016);
        }
        if (keyState['ControlLeft'] || keyState['ControlRight']) {
          input.throttle = Math.max(0, input.throttle - 0.8 * 0.016);
        }
      }
      
      // Store update function globally
      window.updateInput = updateInput;
    }
    
         // Setup camera follow
     function setupCameraFollow() {
       camRig = new THREE.Group();
       camTarget = new THREE.Object3D();
       
       planePivot.add(camRig);
       planePivot.add(camTarget);
       
       // Camera positioned very close to the plane
       camTarget.position.set(0, 0, 0);     // Look at the plane itself
       camRig.position.set(0, 0, -1.5);    // Camera very close to the plane
       camRig.add(camera);
     }
    
    // Load assets
    async function loadAssets() {
      const loader = new GLTFLoader();
      
      try {
        // Load scene
        const sceneGLTF = await loader.loadAsync('scene.glb');
        const sceneModel = sceneGLTF.scene;
        sceneModel.scale.setScalar(10);
        scene.add(sceneModel);
        updateLoadingProgress();
        
        // Load all planes
        for (let i = 0; i < planeFiles.length; i++) {
          try {
            const planeGLTF = await loader.loadAsync(planeFiles[i]);
            const planeModel = planeGLTF.scene;
            
                         // Setup plane model
             planeModel.traverse((child) => {
               if (child.isMesh) {
                 // No shadow settings for better performance
               }
             });
            
            // Store loaded plane
            if (i === 0) {
              // First plane becomes active
              setupPlane(planeModel, planeGLTF);
            }
            
            updateLoadingProgress();
            
          } catch (e) {
            console.warn(`Failed to load ${planeFiles[i]}:`, e);
            updateLoadingProgress();
          }
        }
        
      } catch (e) {
        console.error('Failed to load assets:', e);
        updateLoadingProgress();
      }
    }
    
    // Setup plane
    function setupPlane(planeModel, planeGLTF) {
      // Remove current plane
      if (currentPlane) {
        planePivot.remove(currentPlane);
        if (currentPlaneMixer) {
          mixers.delete(currentPlaneMixer);
          currentPlaneMixer = null;
        }
      }
      
      // Add new plane
      currentPlane = planeModel;
      planePivot.add(currentPlane);
      
      // Setup animations
      if (planeGLTF.animations && planeGLTF.animations.length > 0) {
        currentPlaneMixer = new THREE.AnimationMixer(currentPlane);
        const action = currentPlaneMixer.clipAction(planeGLTF.animations[0]);
        action.play();
        mixers.add(currentPlaneMixer);
      }
      
      // Update UI
      updatePlaneUI();
    }
    
    // Switch plane
    function switchPlane(planeIndex) {
      if (planeIndex === currentPlaneIndex || !gameLoaded) return;
      
      currentPlaneIndex = planeIndex;
      
      // Load the new plane
      const loader = new GLTFLoader();
      loader.loadAsync(planeFiles[planeIndex])
        .then((planeGLTF) => {
          setupPlane(planeGLTF.scene, planeGLTF);
        })
        .catch((e) => {
          console.error(`Failed to load ${planeFiles[planeIndex]}:`, e);
        });
    }
    
    // Update plane UI
    function updatePlaneUI() {
      // Remove active class from all
      document.querySelectorAll('.plane-key').forEach(key => key.classList.remove('active'));
      
      // Add active class to current
      const currentKey = document.getElementById(`plane${currentPlaneIndex + 1}`);
      if (currentKey) {
        currentKey.classList.add('active');
      }
    }
    
         // Reset plane position
     function resetPlanePosition() {
       if (!flight) return; // Don't reset if flight physics not initialized yet
       planePivot.position.set(0, 10, 0);
       planePivot.rotation.set(0, 0, 0);
       flight.speed = THREE.MathUtils.lerp(flight.speedMin, flight.speedMax, input.throttle);
     }
    
    // Window resize handler
    function onWindowResize() {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    }
    
    // Game loop
    function gameLoop() {
      if (!gameLoaded) {
        requestAnimationFrame(gameLoop);
        return;
      }
      
      const deltaTime = 0.016; // Fixed time step for consistent physics
      
      // Update input
      if (window.updateInput) window.updateInput();
      
      // Update flight physics
      updateFlightPhysics(deltaTime);
      
      // Update camera
      updateCamera(deltaTime);
      
      // Update animations
      mixers.forEach(mixer => mixer.update(deltaTime));
      
      // Update HUD
      updateHUD();
      
      // Render
      renderer.render(scene, camera);
      
      requestAnimationFrame(gameLoop);
    }

    // Update camera
    function updateCamera(deltaTime) {
      // Smooth camera follow
      const desiredPosition = camRig.position.clone().applyQuaternion(planePivot.quaternion).add(planePivot.position);
      camera.position.lerp(desiredPosition, 1 - Math.exp(-6 * deltaTime));
      
      const lookAtTarget = camTarget.position.clone().applyQuaternion(planePivot.quaternion).add(planePivot.position);
      camera.lookAt(lookAtTarget);
    }
    
    // Update HUD
    function updateHUD() {
      // Speed
      document.getElementById('speedValue').textContent = Math.round(flight.speed) + ' km/h';
      
      // Altitude
      document.getElementById('altitudeValue').textContent = Math.round(planePivot.position.y) + ' m';
      
      // Throttle
      document.getElementById('throttleValue').textContent = Math.round(input.throttle * 100) + '%';
      
      // FPS (simplified)
      const fps = Math.round(1000 / 16.67); // Approximate
      document.getElementById('fpsValue').textContent = fps;
    }
    
    // Start loading
    loadThreeJS();
  </script>
